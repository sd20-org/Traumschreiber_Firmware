# Traumschreiber TS2607 - Einstellungen (fw 0.0.94)
# Format SCHLUESSEL=WERT, Zeilen mit # sind Kommentare.
# Neustart uebernimmt Aenderungen. Datei unter 2 KB halten
# (die Firmware bearbeitet sie in einem 2048-B-Puffer).

# Funk an (1) oder aus (0)
BLE_ENABLED=1

# Abtastrate in Hz
SAMPLE_RATE=250

# EEG-Verstaerkung fuer alle Kanaele (1,2,4,8,...,128).
# x4 = +-300 mV Reserve (schweissfest), 24-bit BDF verliert keine Aufloesung.

# Kanaele, die per BLE uebertragen werden

# Startton in SOUNDS/ und seine Lautstaerke in Prozent.
# TSBOOT.WAV ist auf 60 abgestimmt; die Firmware begrenzt jede
# Wiedergabe auf -16 dBFS Spitze (Schutz von Versorgung und SD-Karte).
START_SOUND=TSBOOT.WAV
START_SOUND_VOLUME=60

# Lautstaerke fuer Toene aus der App in Prozent (100 = unveraendert)
FIRMWARE_MAX_VOLUME_PERCENT=300

# Debug-Ausgabe ueber RTT (1) und Ausfuehrlichkeit (0-4)
RTT_ENABLED=1
LOG_LEVEL=3

# Taster. Ohne Eintrag gelten Test-Aktionen der Firmware,
# deshalb alle Gesten ausdruecklich belegen.
